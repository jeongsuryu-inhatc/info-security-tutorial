node -e '
const c=require("crypto");
const key=c.randomBytes(32), iv=c.randomBytes(12);
const en=c.createCipheriv("aes-256-gcm",key,iv);

### 과제 1. 비밀번호 해싱 마이그레이션
`shopguard.sqlite`의 **평문 비밀번호를 bcrypt 해시로 일괄 변환**하고(멱등하게),
로그인 검증을 `bcrypt.compareSync`로 바꾼다. cost 10 vs 12 **해싱 시간 차이**도 측정한다.

**Step 1-1. 스크립트를 앱 폴더에 배치하고 bcrypt를 설치한다.**
`week07/migrate-bcrypt.js`를 제출본으로 먼저 작성/복사한 뒤 앱 폴더에 넣는다(그래야 `./db` 가 잡힌다).
```bash
cd starter/shopguard-app
npm install bcrypt
cp ../../week07/migrate-bcrypt.js .        # 제출본을 앱 폴더로
```

**Step 1-2. `migrate-bcrypt.js` 핵심 로직**(그대로 복사·붙여넣기, 전체는 `week07/migrate-bcrypt.js` 참고).
멱등성의 핵심은 **이미 `$2a$/$2b$/$2y$` 로 시작하는 해시는 건너뛰는 것**이다.
```js
const bcrypt = require('bcrypt');
const { openDb } = require('./db');
const BCRYPT_RE = /^\$2[aby]\$/;   // 이미 bcrypt 해시면 재변환하지 않음(멱등)
const COST = 12;

function migrate() {
  const db = openDb();
  try {
    const users = db.prepare('SELECT id, email, password FROM users').all();
    const update = db.prepare('UPDATE users SET password = ? WHERE id = ?');
    const runAll = db.transaction((rows) => {
      for (const u of rows) {
        if (BCRYPT_RE.test(u.password)) continue;          // 이미 해시 → skip
        update.run(bcrypt.hashSync(u.password, COST), u.id); // 평문 → 해시
      }
    });
    runAll(users);
  } finally { db.close(); }
}
migrate();
```

**Step 1-3. `auth.js`의 로그인 검증을 bcrypt로 교체한다**(4주차 Prepared Statement 유지).
평문 비교 `password = ?` 를 없애고, 이메일로 행을 찾은 뒤 **해시를 `compareSync`로 대조**한다.
```js
// auth.js (안전 버전)
const bcrypt = require('bcrypt');
const { openDb } = require('./db');

function login(email, password) {
  const db = openDb();
  try {
    // ✅ Prepared Statement: email 만 바인딩(값으로만 취급 → SQLi 불가)
    const row = db.prepare(
      'SELECT id, email, role, password FROM users WHERE email = ?'
    ).get(email);
    if (!row) return null;
    // ✅ 저장된 bcrypt 해시와 입력 비밀번호를 상수시간 비교
    if (!bcrypt.compareSync(password, row.password)) return null;
    return { id: row.id, email: row.email, role: row.role }; // 해시는 반환하지 않음
  } finally { db.close(); }
}
module.exports = { login };
```

### 과제 2. 민감정보 필드 암호화
전화번호를 **AES-256-GCM**으로 암복호화한다. 키는 **하드코딩 금지 → 환경변수**로 주입,
IV는 매 암호화마다 랜덤, 저장 포맷은 `iv:tag:ciphertext`(base64) 한 문자열로 한다.

**Step 2-1. `aes.js` 전체**(그대로 복사·붙여넣기 → `starter/shopguard-app/aes.js`. 제출본은 `week07/aes.js`).
```js
// aes.js — 전화번호 등 민감정보 AES-256-GCM 암복호화
const crypto = require('crypto');
const ALGO = 'aes-256-gcm';
const IV_BYTES = 12;   // GCM 권장 nonce 길이
const KEY_BYTES = 32;  // AES-256

// 키는 환경변수 SHOPGUARD_KEY(hex 64자 = 32바이트)로만 주입. 하드코딩 금지.
function loadKey() {
  const hex = process.env.SHOPGUARD_KEY;
  if (!hex) throw new Error('SHOPGUARD_KEY 환경변수가 없습니다.');
  const key = Buffer.from(hex, 'hex');
  if (key.length !== KEY_BYTES) throw new Error(`키는 ${KEY_BYTES}바이트여야 합니다.`);
  return key;
}

function encrypt(plaintext) {
  const key = loadKey();
  const iv = crypto.randomBytes(IV_BYTES);            // ✅ 매번 랜덤 IV(재사용 금지)
  const cipher = crypto.createCipheriv(ALGO, key, iv);
  const ct = Buffer.concat([cipher.update(String(plaintext), 'utf8'), cipher.final()]);
  const tag = cipher.getAuthTag();                    // ✅ 무결성 태그
  return [iv.toString('base64'), tag.toString('base64'), ct.toString('base64')].join(':');
}

function decrypt(packed) {
  const key = loadKey();
  const [ivB64, tagB64, ctB64] = String(packed).split(':');
  const decipher = crypto.createDecipheriv(ALGO, key, Buffer.from(ivB64, 'base64'));
  decipher.setAuthTag(Buffer.from(tagB64, 'base64')); // 변조 시 final()에서 예외
  return Buffer.concat([decipher.update(Buffer.from(ctB64, 'base64')), decipher.final()]).toString('utf8');
}

module.exports = { encrypt, decrypt };
```

**Step 2-2. 실행·검증**(왕복 + IV 랜덤성 + 무결성).
```bash
cd starter/shopguard-app
# 32바이트 키를 hex로 생성해 환경변수로 주입(소스/깃에 남기지 않는다)
export SHOPGUARD_KEY=$(node -e "console.log(require('crypto').randomBytes(32).toString('hex'))")

# (a) 암복호화 왕복
node -e "const a=require('./aes'); const b=a.encrypt('010-1234-5678'); console.log(b, '->', a.decrypt(b))"

# (b) 같은 평문이라도 IV가 매번 달라 암호문이 달라짐(결정적이지 않음)
node -e "const a=require('./aes'); console.log(a.encrypt('010-1234-5678')); console.log(a.encrypt('010-1234-5678'))"

# (c) 무결성: 태그 1바이트를 뒤집으면 복호화가 예외로 실패
node -e "const a=require('./aes'); let [iv,t,ct]=a.encrypt('hi').split(':'); const b=Buffer.from(t,'base64'); b[0]^=0xff; try{a.decrypt([iv,b.toString('base64'),ct].join(':'))}catch(e){console.log('변조 감지:',e.message)}"
cd ../..
```
> 기대: (a) `010-1234-5678` 로 왕복 성공, (b) 두 암호문이 서로 다름, (c) `변조 감지: Unsupported
> state or unable to authenticate data` — **GCM 인증 태그가 위·변조를 잡아낸다.**

## 7. 흔한 오류 / 확장 과제

**흔한 오류**
- 비밀번호를 AES로 "암호화 저장" → 잘못. 비밀번호는 **복호화 불가능한 해시**로 저장한다.
- IV를 재사용하거나 고정 → GCM에서 치명적. IV는 매 암호화마다 랜덤.
- 키를 소스/깃에 커밋.

**확장 과제 (심화)**
- bcrypt vs Argon2id를 비교하고 Argon2로 재구현한다.
- 비대칭키(RSA/Ed25519)로 **메시지 서명·검증**을 구현해 무결성·부인방지를 시연한다.
